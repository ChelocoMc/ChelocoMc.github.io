import fitz  # PyMuPDF
import win32com.client as win32

def auditar_stock(pdf_path, excel_path):
    # Extraer datos OCR del PDF
    doc = fitz.open(pdf_path)
    texto = doc[0].get_text()
    datos_pdf = parse_report(texto)

    # Conectar a Excel via COM
    excel = win32.Dispatch('Excel.Application')
    wb = excel.Workbooks.Open(excel_path)
    sheet = wb.Sheets('Stock_Operativo')

    # Comparar y marcar diferencias
    for row in range(2, 1000):
        codigo = sheet.Cells(row, 1).Value
        stock_sistema = sheet.Cells(row, 3).Value
        
        if codigo in datos_pdf:
            if stock_sistema != datos_pdf[codigo]:
                sheet.Cells(row, 4).Value = 'DESCUADRE'
                sheet.Cells(row, 4).Interior.Color = 255 # Red
            else:
                sheet.Cells(row, 4).Value = 'OK'
